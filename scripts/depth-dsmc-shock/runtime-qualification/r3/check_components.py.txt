def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def save(path,value):path.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n')
def load(name,path):
    spec=importlib.util.spec_from_file_location(name,path)
    module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
    return module

def same(a,b):
    if isinstance(a,np.ndarray) or isinstance(b,np.ndarray):
        return isinstance(a,np.ndarray) and isinstance(b,np.ndarray) and a.dtype==b.dtype and a.shape==b.shape and a.tobytes(order='C')==b.tobytes(order='C')
    if isinstance(a,np.generic) or isinstance(b,np.generic):
        return isinstance(a,np.generic) and isinstance(b,np.generic) and a.dtype==b.dtype and a.tobytes()==b.tobytes()
    if isinstance(a,dict) or isinstance(b,dict):
        return isinstance(a,dict) and isinstance(b,dict) and a.keys()==b.keys() and all(same(a[k],b[k]) for k in a)
    if isinstance(a,(tuple,list)) or isinstance(b,(tuple,list)):
        return type(a) is type(b) and len(a)==len(b) and all(same(x,y) for x,y in zip(a,b))
    return type(a) is type(b) and a==b

class RecordedRNG:
    def __init__(self,seed):self.rng=np.random.default_rng(seed);self.calls=[]
    def __getattr__(self,name):
        method=getattr(self.rng,name)
        def call(*args,**kwargs):
            value=method(*args,**kwargs);array=np.asarray(value)
            self.calls.append({'method':name,'args':repr(args),'kwargs':repr(kwargs),'dtype':array.dtype.str,'shape':list(array.shape),'bytes_sha256':hashlib.sha256(array.tobytes()).hexdigest()})
            return value
        return call
    def evidence(self):return {'calls':self.calls,'final_state':self.rng.bit_generator.state}

class SortRecorder:
    def __init__(self):self.calls=[]
    def __getattr__(self,name):return getattr(np,name)
    def argsort(self,keys,*args,**kwargs):
        self.calls.append({'keys':keys.copy(),'args':args,'kwargs':kwargs})
        return np.argsort(keys,*args,**kwargs)

class FixedPermutation:
    def __init__(self,values):self.values=np.asarray(values,dtype=np.int64);self.calls=[]
    def permutation(self,count):
        self.calls.append(count)
        if count!=len(self.values):raise AssertionError('Fixed permutation count')
        return self.values.copy()
    def evidence(self):return {'calls':self.calls,'fixed_permutation':self.values.copy()}

class FixedRNG:
    def __init__(self,values):self.values=np.asarray(values);self.calls=[]
    def random(self,count):
        self.calls.append(count)
        if count!=len(self.values):raise AssertionError('Fixed uniform count')
        return self.values.copy()
    def evidence(self):return {'calls':self.calls}

def captured(function):
    try:return {'value':function(),'error':None}
    except Exception as error:
        return {'value':None,'error':{'type':type(error).__name__,'message':str(error),'position':getattr(error,'position',None),'velocity':getattr(error,'velocity',None)}}

def function_text(path,name):
    text=path.read_text();node=next(n for n in ast.parse(text).body if isinstance(n,(ast.FunctionDef,ast.ClassDef)) and n.name==name)
    return '\n'.join(text.splitlines()[node.lineno-1:node.end_lineno])

            if name!='dsmc.py':check('candidate unchanged '+name,digest(HERE/'candidate/source'/name)==expected)
        for drift,temperature in [(old.U1,1.),(-old.U2,old.TEMP2)]:

                for seed in protocol['seeds']:

                    left,right=captured(lambda:a.sample(r,count)),captured(lambda:b.sample(s,count))
                    check('sample '+label+' '+str(count)+' '+str(seed),left['error'] is None and right['error'] is None and same(left,right) and same(r.evidence(),s.evidence()),evidence={'legacy':left,'candidate':right,'legacy_rng':r.evidence(),'candidate_rng':s.evidence()})
